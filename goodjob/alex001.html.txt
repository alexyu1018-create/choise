class FinMindAPI {
    constructor(token = "") {
        this.baseUrl = "https://api.finmindtrade.com/api/v4/data";
        this.token = token;
    }

    setToken(token) {
        this.token = token;
    }

    async getData(dataset, data_id = "", start_date = "", end_date = "") {
        const params = new URLSearchParams({
            dataset,
            data_id,
            start_date,
            end_date,
            token: this.token
        });

        try {
            const response = await fetch(`${this.baseUrl}?${params.toString()}`);
            if (!response.ok) {
                throw new Error(`HTTP 錯誤! 狀態: ${response.status}`);
            }
            const data = await response.json();
            if (data.status === 200) {
                return data.data;
            } else {
                console.error(`API 錯誤 (${data_id}):`, data.msg);
                updateStatus(`API 錯誤 (${data_id}): ${data.msg}`, true);
                return [];
            }
        } catch (error) {
            console.error(`請求失敗 (${data_id}):`, error);
            updateStatus(`請求失敗 (${data_id}): ${error.message}`, true);
            return [];
        }
    }
}

class StockScreener {
    constructor(api) {
        this.api = api;
        this.stockInfo = [];
    }
    
    async initialize() {
        const stockInfoData = await this.api.getData("TaiwanStockInfo");
        if (stockInfoData.length > 0) {
            this.stockInfo = stockInfoData;
        }
    }

    getStockName(stockId) {
        const stock = this.stockInfo.find(s => s.stock_id === stockId);
        return stock ? stock.stock_name : 'N/A';
    }

    async getStockList() {
        if (this.stockInfo.length === 0) await this.initialize();
        return this.stockInfo
            .filter(s => s.type === 'twse' && !["金融保險業", "建材營造"].includes(s.industry_category))
            .map(s => s.stock_id);
    }

    async calculateCoreProfitMargin(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 730 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);
        
        const df = await this.api.getData("TaiwanStockFinancialStatements", stockId, startDate, endDate);
        
        if (!df || df.length < 4 || !df[0].hasOwnProperty('OperatingIncome')) return [0, {}];

        const sortedDf = df.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 4);

        const quarterlyMargins = [];
        const details = { quarters: [] };

        for (const row of sortedDf) {
            const operatingIncome = row.OperatingIncome || 0;
            const revenue = row.Revenue || 0;
            const taxRate = row.tax_rate || 20;

            if (revenue > 0) {
                const afterTaxProfit = operatingIncome * (1 - taxRate / 100);
                const margin = (afterTaxProfit / revenue) * 100;
                quarterlyMargins.push(margin);
                details.quarters.push({ date: row.date, margin: margin.toFixed(2) });
            }
        }

        if (quarterlyMargins.length > 0) {
            const avgMargin = quarterlyMargins.reduce((a, b) => a + b, 0) / quarterlyMargins.length;
            details.average = avgMargin.toFixed(2);
            return [avgMargin, details];
        }
        
        return [0, {}];
    }
    
    async calculateCoreRoe(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 730 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);

        const [incomeDf, balanceDf] = await Promise.all([
            this.api.getData("TaiwanStockFinancialStatements", stockId, startDate, endDate),
            this.api.getData("TaiwanStockBalanceSheet", stockId, startDate, endDate)
        ]);
        
        if (!incomeDf || !balanceDf || incomeDf.length < 4 || balanceDf.length < 5) return [0, {}];

        const sortedIncome = incomeDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 4);
        const sortedBalance = balanceDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 5);

        const totalOperatingIncome = sortedIncome.reduce((sum, row) => sum + (row.OperatingIncome || 0), 0);
        const taxRates = sortedIncome.map(row => row.tax_rate || 20);
        const avgTaxRate = taxRates.reduce((a, b) => a + b, 0) / taxRates.length;
        
        const totalAfterTaxProfit = totalOperatingIncome * (1 - avgTaxRate / 100);

        const equityValues = sortedBalance.map(row => row.Equity || 0);
        const avgEquity = equityValues.reduce((a, b) => a + b, 0) / equityValues.length;

        if (avgEquity > 0) {
            const coreRoe = (totalAfterTaxProfit / avgEquity) * 100;
            const details = {
                core_roe: coreRoe.toFixed(2),
                total_profit: totalAfterTaxProfit.toFixed(2),
                avg_equity: avgEquity.toFixed(2)
            };
            return [coreRoe, details];
        }

        return [0, {}];
    }

    async calculateMacdWeekly(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 540 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);
        
        const df = await this.api.getData("TaiwanStockPrice", stockId, startDate, endDate);
        
        if (!df || df.length < 100) return [false, {}];

        // Resample to weekly data
        const weeklyData = {};
        df.forEach(row => {
            const date = new Date(row.date);
            const dayOfWeek = date.getDay();
            const friday = new Date(date);
            friday.setDate(date.getDate() - dayOfWeek + 5);
            weeklyData[friday.toISOString().slice(0, 10)] = parseFloat(row.close);
        });

        const weeklyCloses = Object.keys(weeklyData).sort().map(key => weeklyData[key]);

        if (weeklyCloses.length < 35) return [false, {}];
        
        // Calculate MACD
        const calculateEma = (data, span) => {
            const k = 2 / (span + 1);
            let ema = [data[0]];
            for (let i = 1; i < data.length; i++) {
                ema.push(data[i] * k + ema[i - 1] * (1 - k));
            }
            return ema;
        };

        const ema12 = calculateEma(weeklyCloses, 12);
        const ema26 = calculateEma(weeklyCloses, 26);
        const dif = ema12.map((val, i) => val - ema26[i]);
        const dem = calculateEma(dif, 9);
        const osc = dif.map((val, i) => (val - dem[i]) * 2); // FinMind OSC definition

        const latestOsc = osc[osc.length - 1];

        const details = {
            osc: latestOsc.toFixed(4),
            dif: dif[dif.length - 1].toFixed(4),
            dem: dem[dem.length - 1].toFixed(4),
            date: Object.keys(weeklyData).sort().pop()
        };

        return [latestOsc > 0, details];
    }
}

function updateStatus(message, isError = false) {
    statusDiv.textContent = message;
    statusDiv.classList.toggle('text-red-600', isError);
    statusDiv.classList.toggle('text-gray-600', !isError);
}

function toggleUiLock(lock) {
    screeningInProgress = lock;
    startButton.disabled = lock;
    document.getElementById('apiToken').disabled = lock;
    document.getElementById('customStockList').disabled = lock;
    document.querySelectorAll('input[name="screenMode"]').forEach(r => r.disabled = lock);

    buttonLoader.classList.toggle('hidden', !lock);
    buttonText.textContent = lock ? '篩選中...' : '開始篩選';
}

function addResultToTable(stockId, name, margin, roe, osc) {
    const row = document.createElement('tr');
    row.classList.add('hover:bg-gray-50');
    row.innerHTML = `
        <td class="px-6 py-4 whitespace-nowrap text-sm font-medium text-gray-900">${stockId}</td>
        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-500">${name}</td>
        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-500">${margin}</td>
        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-500">${roe}</td>
        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-500 ${osc > 0 ? 'text-green-600' : 'text-red-600'}">${osc}</td>
    `;
    resultsBody.appendChild(row);
}

function updateResultCount(count) {
    resultCountSpan.textContent = `(${count} 筆)`;
}

function exportToCsv() {
    if (finalResults.length === 0) return;
    
    const headers = ['股票代號', '股票名稱', '本業稅後淨利率(%)', '本業ROE(%)', '週MACD_OSC'];
    const rows = finalResults.map(r => [
        r['股票代號'],
        r['股票名稱'],
        r['本業稅後淨利率(%)'],
        r['本業ROE(%)'],
        r['週MACD_OSC']
    ]);
    
    let csvContent = "data:text/csv;charset=utf-8," 
        + headers.join(",") + "\n" 
        + rows.map(e => e.join(",")).join("\n");
        
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement("a");
    link.setAttribute("href", encodedUri);
    const timestamp = new Date().toISOString().slice(0, 19).replace(/[-:T]/g, "");
    link.setAttribute("download", `stock_screening_results_${timestamp}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
}


startButton.addEventListener('click', async () => {
    if (screeningInProgress) return;
    
    const token = document.getElementById('apiToken').value;
    if (!token) {
        updateStatus("請先輸入您的 FinMind API Token。", true);
        return;
    }
    
    toggleUiLock(true);
    resultsBody.innerHTML = '';
    noResultsP.classList.add('hidden');
    exportButton.classList.add('hidden');
    updateResultCount(0);
    finalResults = [];
    
    api.setToken(token);
    
    let stockList;
    const screenMode = document.querySelector('input[name="screenMode"]:checked').value;

    if (screenMode === 'custom') {
        const customText = document.getElementById('customStockList').value.trim();
        if (!customText) {
            updateStatus("請在指定列表模式下輸入股票代碼。", true);
            toggleUiLock(false);
            return;
        }
        stockList = customText.split(/[\s,，\n]+/).filter(Boolean);
    } else {
        updateStatus("正在獲取所有上市公司列表...");
        stockList = await screener.getStockList();
        if(stockList.length === 0) {
            updateStatus("無法獲取股票列表，請檢查 Token 或網路連線。", true);
            toggleUiLock(false);
            return;
        }
    }
    
    updateStatus(`準備篩選 ${stockList.length} 支股票...`);
    await screener.initialize();

    let passedCount = 0;
    for (let i = 0; i < stockList.length; i++) {
        const stockId = stockList[i];
        updateStatus(`處理中: ${stockId} (${i + 1}/${stockList.length})`);

        try {
            const [profitMargin, marginDetails] = await screener.calculateCoreProfitMargin(stockId);
            if (profitMargin <= 10) continue;

            const [coreRoe, roeDetails] = await screener.calculateCoreRoe(stockId);
            if (coreRoe <= 15) continue;
            
            const [macdPass, macdDetails] = await screener.calculateMacdWeekly(stockId);
            if (!macdPass) continue;
            
            // All conditions passed
            passedCount++;
            const stockName = screener.getStockName(stockId);
            addResultToTable(stockId, stockName, marginDetails.average, roeDetails.core_roe, macdDetails.osc);
            finalResults.push({
                '股票代號': stockId,
                '股票名稱': stockName,
                '本業稅後淨利率(%)': marginDetails.average,
                '本業ROE(%)': roeDetails.core_roe,
                '週MACD_OSC': macdDetails.osc,
            });
            updateResultCount(passedCount);

        } catch (error) {
            console.error(`處理 ${stockId} 時發生錯誤:`, error);
        }

        await new Promise(resolve => setTimeout(resolve, 600)); // Rate limit
    }

    if (passedCount === 0) {
        noResultsP.classList.remove('hidden');
    } else {
        exportButton.classList.remove('hidden');
    }

    updateStatus(`篩選完成！共找到 ${passedCount} 支符合條件的股票。`);
    toggleUiLock(false);
});

exportButton.addEventListener('click', exportToCsv);
