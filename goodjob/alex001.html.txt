class FinMindAPI {
    constructor(token = "") {
        this.baseUrl = "https://api.finmindtrade.com/api/v4/data";
        this.token = token;
        // Maximum number of times to retry an API call
        this.MAX_RETRIES = 3; 
    }

    setToken(token) {
        this.token = token;
    }

    // Generalized data fetching with Exponential Backoff
    async getData(dataset, data_id = "", start_date = "", end_date = "") {
        for (let attempt = 0; attempt < this.MAX_RETRIES; attempt++) {
            const params = new URLSearchParams({
                dataset,
                data_id,
                start_date,
                end_date,
                token: this.token
            });

            try {
                const response = await fetch(`${this.baseUrl}?${params.toString()}`);
                
                // Check for rate limit (429) or server error
                if (response.status === 429 || !response.ok) { 
                    const delay = Math.pow(2, attempt) * 1000 + (Math.random() * 1000); // Exponential delay + jitter
                    const statusText = response.status === 429 ? "速率限制 (429)" : `HTTP 錯誤 (${response.status})`;
                    console.warn(`[API Retry] ${statusText} for ${data_id} (${dataset}). Retrying in ${(delay / 1000).toFixed(1)}s... (Attempt ${attempt + 1})`);
                    await new Promise(resolve => setTimeout(resolve, delay));
                    continue; // Go to next attempt
                }

                const data = await response.json();

                if (data.status === 200) {
                    return data.data;
                } else {
                    // Handle specific FinMind API error message
                    if (data.msg && data.msg.includes("data is empty")) return []; // Data not found is not a critical error
                    
                    if (attempt < this.MAX_RETRIES - 1) {
                        const delay = Math.pow(2, attempt) * 1000 + (Math.random() * 1000);
                        console.warn(`[API Retry] FinMind Error (${data_id}, attempt ${attempt + 1}): ${data.msg}. Retrying in ${(delay / 1000).toFixed(1)}s...`);
                        await new Promise(resolve => setTimeout(resolve, delay));
                        continue;
                    } else {
                        console.error(`FinMind API 錯誤 (${data_id}, ${dataset}):`, data.msg);
                        updateStatus(`FinMind API 錯誤 (${data_id}): ${data.msg}`, true);
                        return [];
                    }
                }
            } catch (error) {
                // Handle network errors
                if (attempt < this.MAX_RETRIES - 1) {
                    const delay = Math.pow(2, attempt) * 1000 + (Math.random() * 1000);
                    console.warn(`[Network Retry] 請求失敗 for ${data_id}. Retrying in ${(delay / 1000).toFixed(1)}s... (Attempt ${attempt + 1})`, error);
                    await new Promise(resolve => setTimeout(resolve, delay));
                    continue;
                } else {
                    console.error(`請求失敗 (${data_id}):`, error);
                    updateStatus(`請求失敗 (${data_id}): ${error.message}`, true);
                    return [];
                }
            }
        }
        return [];
    }
}

class StockScreener {
    constructor(api) {
        this.api = api;
        this.stockInfo = [];
    }
    
    async initialize() {
        if (this.stockInfo.length > 0) return;
        const stockInfoData = await this.api.getData("TaiwanStockInfo");
        if (stockInfoData.length > 0) {
            this.stockInfo = stockInfoData;
        }
    }

    getStockName(stockId) {
        const stock = this.stockInfo.find(s => s.stock_id === stockId);
        return stock ? stock.stock_name : 'N/A';
    }

    async getStockList() {
        if (this.stockInfo.length === 0) await this.initialize();
        return this.stockInfo
            .filter(s => s.type === 'twse' && !["金融保險業", "建材營造"].includes(s.industry_category))
            .map(s => s.stock_id);
    }

    // C1: Trailing 4-Quarter Core Net Profit Margin > 10%
    async calculateCoreProfitMargin(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 730 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);
        
        const df = await this.api.getData("TaiwanStockFinancialStatements", stockId, startDate, endDate);
        
        if (!df || df.length < 4 || !df[0].hasOwnProperty('OperatingIncome')) return [0, {}];

        const sortedDf = df.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 4);

        const quarterlyMargins = [];
        const details = { quarters: [] };

        for (const row of sortedDf) {
            const operatingIncome = row.OperatingIncome || 0;
            const revenue = row.Revenue || 0;
            const taxRate = row.tax_rate || 20; 

            if (revenue > 0) {
                const afterTaxProfit = operatingIncome * (1 - taxRate / 100); 
                const margin = (afterTaxProfit / revenue) * 100;
                quarterlyMargins.push(margin);
            }
        }

        if (quarterlyMargins.length === 4) {
            const avgMargin = quarterlyMargins.reduce((a, b) => a + b, 0) / 4;
            details.average = avgMargin.toFixed(2);
            return [avgMargin, details];
        }
        
        return [0, {}];
    }
    
    // C2: Trailing 4-Quarter Core ROE > 15%
    async calculateCoreRoe(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 730 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);

        const [incomeDf, balanceDf] = await Promise.all([
            this.api.getData("TaiwanStockFinancialStatements", stockId, startDate, endDate),
            this.api.getData("TaiwanStockBalanceSheet", stockId, startDate, endDate)
        ]);
        
        if (!incomeDf || !balanceDf || incomeDf.length < 4 || balanceDf.length < 5) return [0, {}];

        const sortedIncome = incomeDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 4);
        const sortedBalance = balanceDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 5);

        const totalOperatingIncome = sortedIncome.reduce((sum, row) => sum + (row.OperatingIncome || 0), 0);
        
        const taxRates = sortedIncome.map(row => row.tax_rate || 20);
        const avgTaxRate = taxRates.length > 0 ? taxRates.reduce((a, b) => a + b, 0) / taxRates.length : 20;
        
        const totalAfterTaxProfit = totalOperatingIncome * (1 - avgTaxRate / 100);

        const equityValues = sortedBalance.map(row => row.Equity || 0);
        const avgEquity = equityValues.reduce((a, b) => a + b, 0) / equityValues.length;

        if (avgEquity > 0) {
            const coreRoe = (totalAfterTaxProfit / avgEquity) * 100;
            const details = { core_roe: coreRoe.toFixed(2) };
            return [coreRoe, details];
        }

        return [0, {}];
    }

    // C3: Weekly MACD OSC > 0
    async calculateMacdWeekly(stockId) {
        const endDate = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 540 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10); 
        
        const df = await this.api.getData("TaiwanStockPrice", stockId, startDate, endDate);
        
        if (!df || df.length < 100) return [false, {}]; 

        // --- Resample Daily to Weekly (using Friday's close as week-end) ---
        const weeklyData = {};
        df.forEach(row => {
            const date = new Date(row.date);
            const dayOfWeek = date.getDay(); // 0=Sun, 5=Fri, 6=Sat
            const friday = new Date(date);
            
            // Logic to find the corresponding Friday's date for the week
            if (dayOfWeek === 6) { friday.setDate(date.getDate() - 1); } 
            else if (dayOfWeek === 0) { friday.setDate(date.getDate() - 2); } 
            else { friday.setDate(date.getDate() - dayOfWeek + 5); }

            const fridayDateKey = friday.toISOString().slice(0, 10);
            
            // Only take the last close price of the week
            if (!weeklyData[fridayDateKey] || date > new Date(weeklyData[fridayDateKey].date)) {
                 weeklyData[fridayDateKey] = { date: row.date, close: parseFloat(row.close) };
            }
        });

        const weeklyCloses = Object.keys(weeklyData)
            .sort()
            .map(key => weeklyData[key].close)
            .filter(c => !isNaN(c)); 

        if (weeklyCloses.length < 35) return [false, {}]; 
        
        // --- Calculate MACD ---
        const calculateEma = (data, span) => {
            const k = 2 / (span + 1);
            let ema = [data[0]];
            for (let i = 1; i < data.length; i++) {
                ema.push(data[i] * k + ema[i - 1] * (1 - k));
            }
            return ema;
        };

        const ema12 = calculateEma(weeklyCloses, 12);
        const ema26 = calculateEma(weeklyCloses, 26);
        const dif = ema12.map((val, i) => val - ema26[i]);
        const dem = calculateEma(dif, 9);
        const osc = dif.map((val, i) => (val - dem[i]) * 2); 

        const latestOsc = osc[osc.length - 1];

        const details = { osc: latestOsc.toFixed(4) };

        return [latestOsc > 0, details]; 
    }
    
    // C4: Check for outstanding Convertible Bonds (CBs)
    async checkConvertibleBond(stockId) {
        const today = new Date().toISOString().slice(0, 10);
        const df = await this.api.getData("TaiwanStockCb", stockId); 
        
        if (!df || df.length === 0) return [false, { status: "無發行" }];

        const outstandingCbs = df.filter(row => {
            if (row.due_date && row.due_date > today) {
                return true;
            }
            return false;
        });

        const status = outstandingCbs.length > 0 ? "流通中" : "已到期/無效";
        return [outstandingCbs.length > 0, { status }];
    }

    // C5: Cumulative Monthly Revenue YoY Growth > Target %
    async calculateCumulativeRevenueYoY(stockId) {
        const today = new Date();
        const endDate = today.toISOString().slice(0, 10);
        const startDate = new Date(today.setFullYear(today.getFullYear() - 3)).toISOString().slice(0, 10); 
        
        const df = await this.api.getData("TaiwanStockMonthRevenue", stockId, startDate, endDate);
        
        if (!df || df.length < 24) return [0, {}];

        // Sort by date descending and take the most recent 24 records
        const sortedDf = df.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 24);

        if (sortedDf.length < 24) return [0, {}];

        // Last 12 months (Current Trailing)
        const currentTrailingRevenue = sortedDf.slice(0, 12).reduce((sum, row) => sum + (row.Revenue || 0), 0);
        
        // Previous 12 months (Previous Trailing)
        const previousTrailingRevenue = sortedDf.slice(12, 24).reduce((sum, row) => sum + (row.Revenue || 0), 0);

        let yoy = 0;
        if (previousTrailingRevenue > 0) {
            yoy = ((currentTrailingRevenue - previousTrailingRevenue) / previousTrailingRevenue) * 100;
        } else if (currentTrailingRevenue > 0) {
            yoy = 9999; // Represents huge growth from near zero base
        }

        const details = { yoy: yoy.toFixed(2) };
        return [yoy, details];
    }
    
    // C6: 5-Year Free Cash Flow Sum > 0
    async calculateFiveYearFCF(stockId) {
        const today = new Date();
        const startDate = new Date(today.setFullYear(today.getFullYear() - 6)).toISOString().slice(0, 10); // 6 years to ensure 20 quarters
        
        const df = await this.api.getData("TaiwanStockCashFlow", stockId, startDate);
        
        if (!df || df.length < 20) return [false, { sum: 0 }];

        const sortedDf = df.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 20);
        if (sortedDf.length < 20) return [false, { sum: 0 }];

        let totalFCF = 0;
        for (const row of sortedDf) {
            const OCF = row.CashFlowsFromUsedInOperatingActivities || 0;
            // CapEx is usually a cash OUTFLOW (negative in FinMind data)
            const CapEx = row.PurchasesOfPropertyPlantAndEquipment || 0; 
            
            // FCF = OCF + CapEx 
            const FCF = OCF + CapEx; 
            totalFCF += FCF;
        }

        const pass = totalFCF > 0;
        return [pass, { sum: totalFCF.toFixed(0) }];
    }

    // C7: 5-Year Average Core ROE > 10%
    async calculateFiveYearAvgRoe(stockId) {
        const today = new Date();
        const startDate = new Date(today.setFullYear(today.getFullYear() - 6)).toISOString().slice(0, 10); // 6 years for 20 quarters

        const [incomeDf, balanceDf] = await Promise.all([
            this.api.getData("TaiwanStockFinancialStatements", stockId, startDate),
            this.api.getData("TaiwanStockBalanceSheet", stockId, startDate)
        ]);
        
        if (!incomeDf || !balanceDf || incomeDf.length < 20 || balanceDf.length < 21) return [0, {}];

        const sortedIncome = incomeDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 20);
        const sortedBalance = balanceDf.sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 21);
        
        if (sortedIncome.length < 20 || sortedBalance.length < 21) return [0, {}];

        const annualCoreRoes = [];
        
        for (let i = 0; i < 20; i += 4) {
            const annualIncome = sortedIncome.slice(i, i + 4);
            const equitySlice = sortedBalance.slice(i, i + 5); 

            if (annualIncome.length === 4 && equitySlice.length === 5) {
                const totalOperatingIncome = annualIncome.reduce((sum, row) => sum + (row.OperatingIncome || 0), 0);
                const avgTaxRate = annualIncome.map(row => row.tax_rate || 20).reduce((a, b) => a + b, 0) / 4;
                const totalAfterTaxProfit = totalOperatingIncome * (1 - avgTaxRate / 100);

                const equityValues = equitySlice.map(row => row.Equity || 0);
                const avgEquity = equityValues.reduce((a, b) => a + b, 0) / equityValues.length;

                if (avgEquity > 0) {
                    const coreRoe = (totalAfterTaxProfit / avgEquity) * 100;
                    annualCoreRoes.push(coreRoe);
                }
            }
        }
        
        if (annualCoreRoes.length === 5) {
            const avgRoe = annualCoreRoes.reduce((a, b) => a + b, 0) / 5;
            const details = { avg_roe: avgRoe.toFixed(2) };
            return [avgRoe, details];
        }

        return [0, {}];
    }
    
    // C8: Branch Broker Activity (Max unique trading branches in last 60 trading days)
    async checkBranchBrokerActivity(stockId, minHouses) {
        const today = new Date().toISOString().slice(0, 10);
        const startDate = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000).toISOString().slice(0, 10);
        
        // This dataset is granular and may be slow.
        const dfBranch = await this.api.getData("TaiwanStockBrokerBranchDailyTrade", stockId, startDate, today);
        
        if (!dfBranch || dfBranch.length === 0) return [false, { max_houses: 0 }];

        const dailyBranchCounts = {};

        // Group by date to find the number of unique branches trading on that day
        dfBranch.forEach(row => {
            const date = row.date;
            if (!dailyBranchCounts[date]) {
                dailyBranchCounts[date] = new Set();
            }
            // Count unique broker branches by their ID
            dailyBranchCounts[date].add(row.broker_branch); 
        });

        let maxHouses = 0;
        let pass = false;
        
        // Find the maximum number of unique branches trading on any single day
        for (const date in dailyBranchCounts) {
            const count = dailyBranchCounts[date].size;
            if (count > maxHouses) {
                maxHouses = count;
            }
            if (count >= minHouses) {
                pass = true;
            }
        }
        
        return [pass, { max_houses: maxHouses }];
    }
}

    statusDiv.textContent = message;
    statusDiv.classList.remove('bg-red-100', 'border-red-200', 'text-red-600', 'bg-indigo-100', 'border-indigo-200', 'text-gray-600', 'bg-green-100', 'border-green-200', 'text-green-700');

function exportToCsv() {
    if (finalResults.length === 0) return;
    
    const headers = [
        '股票代號', '股票名稱', '近四季本業淨利率(%)', '近四季本業ROE(%)', 
        '累計月營收年增率(%)', '週MACD_OSC', '分公司家數峰值', 'CB_狀態'
    ];
    const rows = finalResults.map(r => [
        r['股票代號'],
        r['股票名稱'],
        r['近四季本業淨利率(%)'],
        r['近四季本業ROE(%)'],
        r['累計月營收年增率(%)'],
        r['週MACD_OSC'],
        r['分公司家數峰值'],
        r['CB_狀態']
    ]);
    
    let csvContent = "\ufeff" + headers.join(",") + "\n" + rows.map(e => e.join(",")).join("\n");
    
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);

    const link = document.createElement("a");
    link.setAttribute("href", url);
    const timestamp = new Date().toISOString().slice(0, 16).replace(/[-:T]/g, "");
    link.setAttribute("download", `台股八重篩選結果_${timestamp}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);
}

startButton.addEventListener('click', async () => {
    if (screeningInProgress) return;
    
    const token = apiTokenInput.value;
    if (!token) {
        updateStatus("⚠️ 請先輸入您的 FinMind API Token 才能開始篩選。", true);
        return;
    }
    
    // Get user inputs
    const targetYoY = parseFloat(minRevenueYoYInput.value) || 0;
    const targetMinHouses = parseInt(minBranchBrokersInput.value) || 300;
    
    // Reset UI state
    toggleUiLock(true);
    resultsBody.innerHTML = '';
    noResultsP.classList.add('hidden');
    exportButton.classList.add('hidden');
    updateResultCount(0);
    finalResults = [];
    
    api.setToken(token);
    
    let stockList;
    const screenMode = document.querySelector('input[name="screenMode"]:checked').value;
    const shouldCheckCB = checkCB.checked;

    if (screenMode === 'custom') {
        const customText = document.getElementById('customStockList').value.trim();
        if (!customText) {
            updateStatus("⚠️ 請在指定列表模式下輸入股票代碼。", true);
            toggleUiLock(false);
            return;
        }
        stockList = customText.split(/[\s,，\n]+/)
                              .map(s => s.trim())
                              .filter(Boolean)
                              .filter(s => /^\d+$/.test(s));
    } else {
        updateStatus("🚀 正在獲取所有上市公司列表 (排除金融保險業、建材營造)...");
        try {
             await screener.initialize();
             stockList = await screener.getStockList();
        } catch (e) {
            updateStatus("無法獲取股票列表。請檢查 Token 或網路連線。", true);
            toggleUiLock(false);
            return;
        }

        if(stockList.length === 0) {
            updateStatus("🚫 無法獲取股票列表或列表為空，請檢查 Token 或網路連線。", true);
            toggleUiLock(false);
            return;
        }
    }
    
    updateStatus(`✅ 準備篩選 ${stockList.length} 支股票...`);

    let passedCount = 0;
    const TOTAL_STOCKS = stockList.length;

    for (let i = 0; i < TOTAL_STOCKS; i++) {
        const stockId = stockList[i];
        
        try {
            // --- 1. C1: Profit Margin Check ---
            updateStatus(`🔍 處理中: ${stockId} (1/8 - 獲利能力)`);
            const [profitMargin, marginDetails] = await screener.calculateCoreProfitMargin(stockId);
            if (profitMargin <= 10) continue;

            // --- 2. C2: ROE Check ---
            updateStatus(`🔍 處理中: ${stockId} (2/8 - 經營效率)`);
            const [coreRoe, roeDetails] = await screener.calculateCoreRoe(stockId);
            if (coreRoe <= 15) continue;
            
            // --- 3. C3: MACD Check ---
            updateStatus(`🔍 處理中: ${stockId} (3/8 - 上漲動能)`);
            const [macdPass, macdDetails] = await screener.calculateMacdWeekly(stockId);
            if (!macdPass) continue;
            
            // --- 4. C5: Cumulative Revenue YoY Check ---
            updateStatus(`🔍 處理中: ${stockId} (4/8 - 累計營收動能)`);
            const [cumYoY, cumYoYDetails] = await screener.calculateCumulativeRevenueYoY(stockId);
            if (cumYoY < targetYoY) continue;

            // --- 5. C6: 5-Year FCF Check ---
            updateStatus(`🔍 處理中: ${stockId} (5/8 - 長期自由金流)`);
            const [fcfPass, fcfDetails] = await screener.calculateFiveYearFCF(stockId);
            if (!fcfPass) continue;

            // --- 6. C7: 5-Year Average ROE Check ---
            updateStatus(`🔍 處理中: ${stockId} (6/8 - 長期平均ROE)`);
            const [avgRoe, avgRoeDetails] = await screener.calculateFiveYearAvgRoe(stockId);
            if (avgRoe < 10) continue; 

            // --- 7. C8: Branch Broker Activity Check ---
            updateStatus(`🔍 處理中: ${stockId} (7/8 - 分公司交易家數)`);
            const [brokerPass, brokerDetails] = await screener.checkBranchBrokerActivity(stockId, targetMinHouses);
            if (!brokerPass) continue;
            
            // --- 8. C4: Convertible Bond Check (Optional) ---
            let cbDetails = { status: "未篩選" }; 

            if (shouldCheckCB) {
                updateStatus(`🔍 處理中: ${stockId} (8/8 - 可轉債狀態)`);
                const [cbPass, details] = await screener.checkConvertibleBond(stockId);
                cbDetails = details;
                
                if (!cbPass) continue;
            }
            
            // All conditions passed (or optionally passed)
            passedCount++;
            const stockName = screener.getStockName(stockId);
            const oscValue = parseFloat(macdDetails.osc);

            addResultToTable(
                stockId, 
                stockName, 
                marginDetails.average, 
                roeDetails.core_roe, 
                cumYoYDetails.yoy,
                oscValue.toFixed(4),
                brokerDetails.max_houses,
                cbDetails.status
            );

            // Store for CSV export
            finalResults.push({
                '股票代號': stockId,
                '股票名稱': stockName,
                '近四季本業淨利率(%)': marginDetails.average,
                '近四季本業ROE(%)': roeDetails.core_roe,
                '累計月營收年增率(%)': cumYoYDetails.yoy,
                '週MACD_OSC': oscValue.toFixed(4),
                '分公司家數峰值': brokerDetails.max_houses,
                'CB_狀態': cbDetails.status,
            });
            updateResultCount(passedCount);

        } catch (error) {
            console.error(`處理 ${stockId} 時發生未預期錯誤:`, error);
        }

        // Add a deliberate slight delay between stock checks to respect overall API limits
        await new Promise(resolve => setTimeout(resolve, 300)); 
    }

    if (passedCount === 0) {
        noResultsP.classList.remove('hidden');
    } else {
        exportButton.classList.remove('hidden');
    }

    const filterCount = shouldCheckCB ? '八重' : '七重';
    updateStatus(`🎉 篩選完成！共找到 ${passedCount} 支符合${filterCount}條件的股票。`);
    toggleUiLock(false);
});
